## 2. New Tables (PROPOSED)

### 2.1 incidents

```sql
-- PROPOSED — ยังไม่ได้สร้าง
CREATE TABLE incidents (
  id              UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
  public_id       TEXT        NOT NULL,
  title           TEXT        NOT NULL,
  description     TEXT,
  
  -- Classification
  category_id     INTEGER     REFERENCES categories(id),
  subcategory_id  INTEGER     REFERENCES subcategories(id),
  location        TEXT,
  
  -- Status
  status          TEXT        NOT NULL DEFAULT 'open'
                              CHECK (status IN ('open', 'in_progress', 'resolved', 'closed')),
  priority        TEXT        NOT NULL DEFAULT 'normal'
                              CHECK (priority IN ('low', 'normal', 'high', 'critical')),
  
  -- Assignment
  created_by      UUID        REFERENCES auth.users(id),
  assigned_to     UUID        REFERENCES auth.users(id),
  
  -- Denormalized counter
  report_count    INTEGER     NOT NULL DEFAULT 0,
  
  -- Notification tracking
  last_notified_at          TIMESTAMPTZ,
  last_notification_count   INTEGER     DEFAULT 0,
  
  -- Timestamps
  created_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  resolved_at     TIMESTAMPTZ,
  closed_at       TIMESTAMPTZ
);

-- Constraints
ALTER TABLE incidents ADD CONSTRAINT incidents_public_id_unique UNIQUE (public_id);

-- Indexes
CREATE INDEX idx_incidents_status ON incidents(status);
CREATE INDEX idx_incidents_category_id ON incidents(category_id);
CREATE INDEX idx_incidents_created_at ON incidents(created_at DESC);
CREATE INDEX idx_incidents_assigned_to ON incidents(assigned_to);
```

**Fields explanation:**

| Column | Purpose | Notes |
|--------|---------|-------|
| public_id | Human-readable ID | Format: INC-YYYYMMDD-NNN, generated by DB sequence |
| report_count | จำนวน reports ที่ active | Maintained by trigger on incident_reports |
| last_notified_at | เวลาที่ส่ง LINE ล่าสุด | ใช้สำหรับ debounce calculation |
| last_notification_count | จำนวน reports ตอนส่ง notif ล่าสุด | ใช้คำนวณ "เพิ่ม X รายตั้งแต่ครั้งก่อน" |

**Status transitions:**

```
open → in_progress → resolved → closed
         ↑                        │
         └────────────────────────┘  (reopen)
```

### 2.2 incident_reports

```sql
-- PROPOSED — ยังไม่ได้สร้าง
CREATE TABLE incident_reports (
  id              UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
  incident_id     UUID        NOT NULL REFERENCES incidents(id) ON DELETE CASCADE,
  report_id       UUID        NOT NULL REFERENCES reports(id) ON DELETE CASCADE,
  
  -- Link metadata
  linked_by       UUID        REFERENCES auth.users(id),
  linked_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  link_source     TEXT        NOT NULL DEFAULT 'manual'
                              CHECK (link_source IN ('manual', 'suggestion_accepted', 'auto')),
  
  -- Soft remove
  removed_by      UUID        REFERENCES auth.users(id),
  removed_at      TIMESTAMPTZ,
  removed_reason  TEXT,
  is_active       BOOLEAN     NOT NULL DEFAULT TRUE,
  
  -- Prevent duplicate links
  CONSTRAINT uq_incident_report UNIQUE (incident_id, report_id)
);

-- Indexes (partial for active records)
CREATE INDEX idx_ir_incident_active ON incident_reports(incident_id) WHERE is_active = TRUE;
CREATE INDEX idx_ir_report_active ON incident_reports(report_id) WHERE is_active = TRUE;
```

**Key behaviors:**

| Scenario | Action | is_active |
|----------|--------|-----------|
| Link report to incident | INSERT | TRUE |
| Remove report from incident | UPDATE | FALSE |
| Re-link removed report | UPDATE back | TRUE |

> **Critical:** `ON DELETE CASCADE` เฉพาะ incident → ถ้าลบ incident ก็ลบ links  
> Report ไม่ถูกลบ ยังอยู่ในระบบเสมอ

### 2.3 incident_suggestions

CREATE TABLE incident_suggestions (

  report_id       UUID        NOT NULL REFERENCES reports(id) ON DELETE CASCADE,

### 2.5 notification_queue

```sql
-- PROPOSED — ยังไม่ได้สร้าง
CREATE TABLE notification_queue (
  id                UUID        PRIMARY KEY DEFAULT gen_random_uuid(),
  
  -- Context
  notification_type TEXT        NOT NULL
                                CHECK (notification_type IN (
                                  'new_report', 'incident_created', 'incident_update',
                                  'incident_alert', 'status_changed', 'bulk_reply',
                                  'incident_closed'
                                )),
  report_id         UUID        REFERENCES reports(id) ON DELETE SET NULL,
  incident_id       UUID        REFERENCES incidents(id) ON DELETE SET NULL,
  bulk_action_id    UUID        REFERENCES bulk_actions(id) ON DELETE SET NULL,
  
  -- Target
  line_group_id     TEXT        NOT NULL,
  
  -- Message
  payload           JSONB       NOT NULL,
  
  -- Debounce
  debounce_key      TEXT,
  send_after        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  
  -- Status
  status            TEXT        NOT NULL DEFAULT 'pending'
                                CHECK (status IN ('pending', 'processing', 'sent', 'failed', 'cancelled')),
  attempt_count     INTEGER     NOT NULL DEFAULT 0,
  max_attempts      INTEGER     NOT NULL DEFAULT 3,
  last_error        TEXT,
  
  -- Idempotency
  idempotency_key   TEXT,
  
  -- Timestamps
  created_at        TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  processed_at      TIMESTAMPTZ,
  sent_at           TIMESTAMPTZ
);

-- Indexes
CREATE INDEX idx_nq_pending ON notification_queue(send_after) WHERE status = 'pending';
CREATE INDEX idx_nq_debounce ON notification_queue(debounce_key) WHERE debounce_key IS NOT NULL AND status = 'pending';
CREATE UNIQUE INDEX idx_nq_idempotency ON notification_queue(idempotency_key) WHERE idempotency_key IS NOT NULL;
```

## 3. Existing Table Modifications (PROPOSED CHANGE)

### 3.1 report_logs — เพิ่ม bulk_action_id

```sql
-- PROPOSED CHANGE — ยังไม่ได้รัน
ALTER TABLE report_logs 
  ADD COLUMN IF NOT EXISTS bulk_action_id UUID REFERENCES bulk_actions(id);

CREATE INDEX idx_rl_bulk_action ON report_logs(bulk_action_id) WHERE bulk_action_id IS NOT NULL;
```

**เหตุผล:** เชื่อมว่า report_log นี้เกิดจาก Bulk Action ไหน  
**ผลกระทบ:** Column เป็น nullable → ไม่กระทบ record เดิม  
**ทางเลือก:** ไม่มี — จำเป็นต้องมี linkage  
**ความเสี่ยง:** ต่ำ — เพิ่ม column nullable ไม่กระทบ query เดิม  
**Rollback:** `ALTER TABLE report_logs DROP COLUMN bulk_action_id;`

### 3.2 audit_logs — เพิ่ม bulk_action_id, incident_id

```sql
-- PROPOSED CHANGE — ยังไม่ได้รัน
ALTER TABLE audit_logs 
  ADD COLUMN IF NOT EXISTS bulk_action_id UUID,
  ADD COLUMN IF NOT EXISTS incident_id UUID;
```

**เหตุผล:** Audit trail ต้องเชื่อม action กับ incident/bulk_action  
**ผลกระทบ:** ต่ำ — nullable columns  
**Rollback:** `ALTER TABLE audit_logs DROP COLUMN bulk_action_id, DROP COLUMN incident_id;`

### 3.3 audit_logs RLS — ขยาย INSERT policy

```sql
-- PROPOSED CHANGE — ยังไม่ได้รัน

-- ปัจจุบัน: INSERT เฉพาะ super_admin
-- เปลี่ยนเป็น: INSERT ให้ staff ทุกคน (เพื่อ log bulk actions)

DROP POLICY IF EXISTS "Super Admins can insert audit logs" ON audit_logs;
CREATE POLICY "Staff can insert audit logs"
  ON audit_logs FOR INSERT WITH CHECK (
    EXISTS (SELECT 1 FROM staff_users WHERE auth_user_id = auth.uid())
  );
```

**เหตุผล:** Bulk action ทำโดย staff/manager ต้อง log ได้  
**ผลกระทบ:** ขยายสิทธิ์ INSERT — ต้องตรวจว่า SELECT ยังจำกัด  
**ความเสี่ยง:** ปานกลาง — ต้อง review ว่า SELECT policy ยังจำกัดอยู่  
**Rollback:** Restore original policy name + condition

## 4. Triggers (PROPOSED)

### 4.1 Auto-update incidents.updated_at

```sql
-- PROPOSED
CREATE OR REPLACE FUNCTION trigger_set_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_incidents_updated_at
  BEFORE UPDATE ON incidents
  FOR EACH ROW EXECUTE FUNCTION trigger_set_updated_at();
```

### 4.2 Sync incidents.report_count

```sql
-- PROPOSED
CREATE OR REPLACE FUNCTION sync_incident_report_count()
RETURNS TRIGGER AS $$
DECLARE
  target_incident_id UUID;
BEGIN
  target_incident_id := COALESCE(NEW.incident_id, OLD.incident_id);
  
  UPDATE incidents 
  SET report_count = (
    SELECT COUNT(*) FROM incident_reports 
    WHERE incident_id = target_incident_id AND is_active = TRUE
  ),
  updated_at = NOW()
  WHERE id = target_incident_id;
  
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_sync_report_count
  AFTER INSERT OR UPDATE OF is_active ON incident_reports
  FOR EACH ROW EXECUTE FUNCTION sync_incident_report_count();
```

## 5. Functions (PROPOSED)

### 5.1 Generate Incident Public ID

```sql
-- PROPOSED
CREATE SEQUENCE IF NOT EXISTS incident_daily_seq;

CREATE OR REPLACE FUNCTION generate_incident_public_id()
RETURNS TEXT AS $$
BEGIN
  RETURN 'INC-' || TO_CHAR(NOW(), 'YYYYMMDD') || '-' || 
         LPAD(NEXTVAL('incident_daily_seq')::TEXT, 3, '0');
END;
$$ LANGUAGE plpgsql;
```

> **Note:** Sequence resets ต้องจัดการด้วย cron หรือ logic ใน application

### 5.2 Bulk Update Reports (Atomic)

```sql
-- PROPOSED
CREATE OR REPLACE FUNCTION bulk_update_reports(
  p_report_ids   UUID[],
  p_new_status   TEXT,
  p_message      TEXT,
  p_staff_id     UUID,
  p_bulk_action_id UUID
) RETURNS JSONB AS $$
DECLARE
  v_report_id UUID;
  v_old_status TEXT;
  v_success INT := 0;
  v_failed INT := 0;
BEGIN
  FOREACH v_report_id IN ARRAY p_report_ids LOOP
    BEGIN
      SELECT status INTO v_old_status FROM reports WHERE id = v_report_id;
      
      UPDATE reports 
      SET status = p_new_status, updated_at = NOW()
      WHERE id = v_report_id;
      
      INSERT INTO report_logs (report_id, user_id, action, old_status, new_status, remark, bulk_action_id)
      VALUES (v_report_id, p_staff_id, 'status_updated', v_old_status, p_new_status, p_message, p_bulk_action_id);
      
      v_success := v_success + 1;
    EXCEPTION WHEN OTHERS THEN
      v_failed := v_failed + 1;
    END;
  END LOOP;
  
  RETURN jsonb_build_object('success', v_success, 'failed', v_failed);
END;
$$ LANGUAGE plpgsql;
```
